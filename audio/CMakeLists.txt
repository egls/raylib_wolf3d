cmake_minimum_required(VERSION 3.16)
project(wolf_audio LANGUAGES C)
option(SD_AUDIO_SPEAR "Use Spear 560 Hz music timer" OFF)
add_library(wolf_audio sd_audio.c vendor/opl3.c)
target_include_directories(wolf_audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(wolf_audio PUBLIC c_std_11)
if(SD_AUDIO_SPEAR)
  target_compile_definitions(wolf_audio PRIVATE SD_AUDIO_SPEAR)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(wolf_audio PRIVATE -Wall -Wextra -Wpedantic -fsigned-char)
endif()
add_executable(wolf_audio_render render.c)
target_link_libraries(wolf_audio_render PRIVATE wolf_audio)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(wolf_audio_render PRIVATE -Wall -Wextra -Wpedantic -fsigned-char)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(wolf_audio_test tests/test_audio.c)
  target_link_libraries(wolf_audio_test PRIVATE wolf_audio)
  add_test(NAME audio_contract COMMAND wolf_audio_test)
endif()
